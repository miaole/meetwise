  for (const p of libFiles) {
    const src = execFileSync('git', ['show', `${prefixRef}:${p}`], { cwd: root, encoding: 'utf8' });

gate('[5] <detailsx> => PASS', parseReviewFileVerdict(NEAR_MISS), 'PASS');
